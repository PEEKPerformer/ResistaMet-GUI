# ResistaMet-GUI dependencies
# Install with: pip install -r requirements.txt

# GUI framework. LGPL-3.0 (official Qt Company binding) — chosen over PyQt5
# (GPL-3.0) so MIT-licensed downstreams can distribute the PyInstaller .exe
# without GPL contamination.
PySide6>=6.5.0

# Instrument communication
pyvisa>=1.11.0
pyvisa-py>=0.5.0

# Data processing and visualization
numpy>=1.19.0
matplotlib>=3.3.0
# Live time-series tabs (Resistance, V-source, I-source, 4PP results) use
# pyqtgraph for smooth high-rate updates; matplotlib stays for the 4PP
# histogram and I-V sweep canvases.
pyqtgraph>=0.13.0

# HDF5 export backend. Bundled here (rather than in optional-dependencies
# only) so the PyInstaller .exe ships with HDF5 support enabled; otherwise
# the Output → HDF5 radio is permanently grayed out for .exe users with
# no pip-install escape hatch.
h5py>=3.0.0

# Typed settings / event contract (resistamet_gui/schema). Required by the
# frozen build too: --self-test imports the schema, and the headless session
# and API layer are built on these models.
pydantic>=2.0

# Localhost API for the headless session (resistamet_gui/api). Optional for
# the GUI; required by the sidecar the Tauri shell will launch.
fastapi>=0.110
uvicorn>=0.27
# uvicorn speaks WebSocket only through a separately installed implementation;
# without one it answers every upgrade with 404 and the desktop UI shows
# "Reconnecting" forever. Found on the lab PC in the first frozen build.
websockets>=13

# User-space driver for NI GPIB-USB adapters (resistamet_gui/gpib_usb), the
# GPIB route on a Mac where NI ships no driver. pyusb is BSD; it needs a
# libusb shared library at runtime (brew install libusb / apt install
# libusb-1.0-0; bundled into the frozen backend on macOS by
# resistamet-api.spec). Without libusb the driver reports itself
# unavailable and nothing else changes.
pyusb>=1.2

# Testing (development)
pytest>=7.0.0
pytest-cov>=4.0.0
